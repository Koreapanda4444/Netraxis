cmake_minimum_required(VERSION 3.24)
project(Netraxis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
if(MSVC)
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Sql)
add_library(netraxis_domain STATIC src/domain/NetworkValues.cpp)
target_include_directories(netraxis_domain PUBLIC src)
target_link_libraries(netraxis_domain PUBLIC Qt6::Core Qt6::Network)
add_library(netraxis_core STATIC src/core/Paths.cpp src/core/Logger.cpp src/core/TaskManager.cpp src/core/TaskManager.h src/core/Settings.cpp)
target_link_libraries(netraxis_core PUBLIC netraxis_domain)
add_library(netraxis_platform STATIC src/platform/SystemInfo.cpp)
target_link_libraries(netraxis_platform PUBLIC netraxis_domain)
add_library(netraxis_routers STATIC src/routers/ManagementEndpoint.cpp)
target_link_libraries(netraxis_routers PUBLIC netraxis_domain)
add_library(netraxis_ui STATIC src/ui/MainWindow.cpp src/ui/MainWindow.h src/ui/StatusSurface.cpp src/ui/StatusSurface.h)
target_link_libraries(netraxis_ui PUBLIC netraxis_core netraxis_platform Qt6::Widgets)
qt_add_executable(netraxis src/app/main.cpp src/app/ApplicationLifecycle.cpp src/app/ApplicationLifecycle.h)
target_link_libraries(netraxis PRIVATE netraxis_ui netraxis_core netraxis_platform netraxis_routers)
set_target_properties(netraxis PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
install(TARGETS netraxis BUNDLE DESTINATION . RUNTIME DESTINATION bin)
